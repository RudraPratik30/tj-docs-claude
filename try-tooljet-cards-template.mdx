{/*
  ToolJet Docs — Card Components Reference
  ==========================================
  This file contains reusable card component templates for ToolJet documentation pages.
  Copy the relevant components into any .mdx file where cards are needed.

  REQUIREMENTS:
  - File must be .mdx (not .md)
  - Must import homepage.css (adjust relative path based on file location):
      import '../../homepage.css';
      import Link from '@docusaurus/Link';

  THREE CARD VARIANTS ARE AVAILABLE:
  1. SetupCard  — Large card with icon, title, and description (used on setup/deploy pages)
  2. PluginCard — Compact single-row card with icon and title (used on marketplace pages)
  3. Custom     — Use the base Card component directly for any layout

  ICON LOCATIONS:
  - Setup icons: /img/setup/icons/<iconId>.svg
  - Plugin icons: /img/marketplace/plugins/<pluginId>/icon.svg
  - Add new icons to docs/static/img/ in the appropriate folder
*/}

{/* ================================================================
   BASE CARD COMPONENT (required by all variants)
   - className: additional Tailwind classes
   - href: link destination (internal path or external URL)
   - isExternal: set true for external URLs (opens in new tab)
   ================================================================ */}

import Link from '@docusaurus/Link';

export const Card = ({ className = '', href, children, isExternal }) => {
    const cardContent = (
        <div className={`relative rounded-lg bg-white dark:bg-gray-800 group pt-[0.5px] pb-[3px] shadow-[inset_0_0_0_1.2px_rgba(243,244,246,1)] dark:shadow-[inset_0_0_0_1.2px_rgba(55,65,81,1)] ${className}`}>
            <div className="absolute inset-0 rounded-lg opacity-0 group-hover:opacity-100 transition-opacity duration-50"
                style={{
                    background: 'linear-gradient(to bottom, #3B82F6, transparent) border-box',
                    border: '0px',
                    backgroundClip: 'padding-box, border-box',
                    backgroundOrigin: 'border-box',
                    pointerEvents: 'none'
                }}>
            </div>
            <div className="m-0.5 relative z-10 h-full rounded-lg border border-gray-300 dark:border-gray-700 bg-white dark:bg-gray-900">
                {children}
            </div>
        </div>
    );
    return href ? (
        isExternal ? (
            <a href={href} target="_blank" rel="noopener noreferrer" className="no-underline hover:no-underline block h-full" style={{ textDecoration: 'none' }}>
                {cardContent}
            </a>
        ) : (
            <Link to={href} className="no-underline hover:no-underline block h-full" style={{ textDecoration: 'none' }}>
                {cardContent}
            </Link>
        )
    ) : cardContent;
};

{/* ================================================================
   VARIANT 1: SetupCard (large card with icon + title + description)
   - iconId: filename (without .svg) from /img/setup/icons/
   - title: card heading
   - description: card body text
   - href: link destination
   - isExternal: true for external URLs
   - Height: fixed at 280px
   - Best for: setup pages, overview pages, feature grids
   ================================================================ */}

export const SetupCard = ({ iconId, title, description, href, isExternal }) => (
    <Card href={href} isExternal={isExternal} className="h-[280px] transition-all duration-300 ease-in-out hover:shadow-lg cursor-pointer group relative">
        <div className="p-8 h-full flex flex-col">
            <div className="flex flex-col space-y-1.5">
                <div className="w-12 h-12 mb-5 rounded-lg shadow-md flex items-center justify-center transition-all duration-300 ease-in-out group-hover:shadow-lg bg-gray-100 dark:bg-gray-700">
                    <img
                        src={`/img/setup/icons/${iconId}.svg`}
                        alt={title}
                        style={{ width: '28px', height: '28px', boxShadow: 'none', borderRadius: '0' }}
                    />
                </div>
                <h3 className="text-xl font-semibold leading-none tracking-tight text-black dark:text-white" style={{ textDecoration: 'none' }}>{title}</h3>
            </div>
            <div className="pt-2 flex-grow">
                <p className="text-sm text-gray-600 dark:text-gray-400 line-clamp-5" style={{ textDecoration: 'none' }}>{description}</p>
            </div>
        </div>
    </Card>
);

{/* ================================================================
   VARIANT 2: PluginCard (compact single-row card with icon + title)
   - pluginId: folder name from /img/marketplace/plugins/<pluginId>/
   - title: plugin display name
   - href: link to plugin docs page
   - Height: fixed at 72px
   - Best for: marketplace listings, plugin grids
   ================================================================ */}

export const PluginCard = ({ pluginId, title, href }) => (
    <Card href={href} className="h-[72px] transition-all duration-300 ease-in-out hover:shadow-lg cursor-pointer group relative">
        <div className="p-3 h-full flex items-center justify-between">
            <div className="flex items-center space-x-3">
                <div className="w-10 h-10 rounded-lg shadow flex items-center justify-center bg-gray-100 dark:bg-gray-700">
                    <img
                        src={`/img/marketplace/plugins/${pluginId}/icon.svg`}
                        alt={title}
                        style={{ width: '24px', height: '24px', boxShadow: 'none', borderRadius: '0' }}
                    />
                </div>
                <span className="text-sm font-semibold text-black dark:text-white">{title}</span>
            </div>
        </div>
    </Card>
);

{/* ================================================================
   SECTION WRAPPER (groups cards in a responsive grid)
   - title: optional h2 heading above the grid
   - description: optional subtitle text
   - cols: 2 or 3 (default 3) — controls grid column count
   ================================================================ */}

export const CardSection = ({ title, description, cols = 3, children }) => (
    <div style={{ marginBottom: '3rem', paddingBottom: '2rem', borderBottom: '1px solid #e5e7eb' }}>
        {title && <h2 className="text-2xl font-bold text-black dark:text-white">{title}</h2>}
        {description && <p className="text-sm text-gray-500 dark:text-gray-300 mb-4">{description}</p>}
        <div className={`grid grid-cols-1 ${cols === 2 ? 'md:grid-cols-2' : 'md:grid-cols-2 lg:grid-cols-3'} gap-4`}>
            {children}
        </div>
    </div>
);
